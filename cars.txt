<?php

require_once "db.php";

// GET /cars
function getCars()
{
    $db = dbConnect();
    $cars = $db->query("SELECT * FROM cars")->fetchAll();
    echo json_encode($cars);
}

function addCar()
{
    $body = json_decode(file_get_contents("php://input"), true);

    if (!isset($body["brand"]) || !isset($body["model"]) || !isset($body["type"]) || !isset($body["rate_per_day"]) || !isset($body["transmission_type"]) || !isset($body["gasoline_type"]) || !isset($body["seats"])) {
        http_response_code(400);
        echo json_encode(["error" => "Missing required fields"]);
        return;
    }

    $db = dbConnect();
    $myQuery = $db->prepare("INSERT INTO cars (brand, model, type, rate_per_day, transmission_type, gasoline_type, seats) VALUES (?, ?, ?, ?, ?, ?, ?)");
    $myQuery->execute([
        $body["brand"],
        $body["model"],
        $body["type"],
        $body["rate_per_day"],
        $body["transmission_type"],
        $body["gasoline_type"],
        $body["seats"]
    ]);

    echo json_encode(["message" => "Car added successfully"]);
}

function updateCar($id)
{
    $body = json_decode(file_get_contents("php://input"), true);

    $db = dbConnect();

    $myQuery = $db->prepare("UPDATE cars SET brand = ?, model = ?, type = ?, rate_per_day = ?, transmission_type = ?, gasoline_type = ?, seats = ? WHERE car_id = ?");
    $myQuery->execute([
        $body["brand"],
        $body["model"],
        $body["type"],
        $body["rate_per_day"],
        $body["transmission_type"],
        $body["gasoline_type"],
        $body["seats"],
        $id
    ]);

    if ($myQuery->rowCount() === 0) {
        http_response_code(404);
        echo json_encode(["error" => "Car not found"]);
        return;
    }

    echo json_encode(["message" => "Car updated"]);
}

function deleteCar($id)
{
    $db = dbConnect();

    $myQuery = $db->prepare("DELETE FROM cars WHERE car_id = ?");
    $myQuery->execute([$id]);

    if ($myQuery->rowCount() === 0) {
        http_response_code(404);
        echo json_encode(["error" => "Car not found"]);
        return;
    }

    echo json_encode(["message" => "Car deleted"]);
}
